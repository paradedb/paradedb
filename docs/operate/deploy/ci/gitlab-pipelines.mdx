## Sample GitLab Pipeline

```yaml
paradedb-in-gitlab-pipelines:
  # The list of available tags can be found at https://hub.docker.com/r/paradedb/paradedb/tags
  image: paradedb/paradedb:latest
  services:
    - name: paradedb/paradedb:latest
      alias: paradedb
  variables:
    POSTGRES_USER: testuser
    POSTGRES_DB: testdb
    POSTGRES_HOST_AUTH_METHOD: trust
  script:
    - psql -h "paradedb" -U testuser -d testdb -c "CALL paradedb.create_paradedb_test_table(schema_name => 'public', table_name => 'mock_items');"
    - psql -h "paradedb" -U testuser -d testdb -c "SELECT description, rating, category FROM mock_items LIMIT 3;"
    - psql -h "paradedb" -U testuser -d testdb -c "CREATE INDEX search_idx ON mock_items USING paradedb (id, description, category, rating, in_stock, created_at, metadata, weight_range);"
    - psql -h "paradedb" -U testuser -d testdb -c "SELECT description, rating, category FROM mock_items WHERE description ||| 'running shoes' AND rating > 2 ORDER BY rating LIMIT 5;"
    - psql -h "paradedb" -U testuser -d testdb -c "SELECT description, rating, category FROM mock_items WHERE description &&& 'running shoes' AND rating > 2 ORDER BY rating LIMIT 5;"
```
